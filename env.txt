# ZTXBAS configuration example.
# Copy to .env for local dev, or set as environment variables in production.

# --- Required (set these; no safe defaults) ---
# ZTXBAS_DATA_DIR — where the SQLite DB and JWT signing key live.
# Leave unset when running the container image (defaults to
# /var/lib/ztxbas, matches the volume mount). Set explicitly only when
# running the binary directly, not in Docker.
# ZTXBAS_DATA_DIR=./data

# ZTXBAS_PUBLIC_URL — public address the RP, operator, and enrolled
# mobile devices all reach the server at. Baked into enrollment links,
# QR codes, and JWT `iss` claims. Loopback (127.0.0.1) will break
# mobile enrollment — always use a routable host address.
# ZTXBAS_PUBLIC_URL=https://ztxbas.example.com

ZTXBAS_SMTP_HOST=
ZTXBAS_SMTP_PORT=
ZTXBAS_SMTP_USER=
ZTXBAS_SMTP_PASS=
ZTXBAS_SMTP_FROM=

# --- Optional networking ---
# Main HMAC-authed API listener (RP-facing).
# ZTXBAS_LISTEN_ADDR=:8443

# Admin console listener. Loopback by default — see the fail-safe below.
# In Docker, must be 0.0.0.0:8080 (with ALLOW_INSECURE_CONSOLE=1) even
# when the host-side port maps to loopback, because Docker's port
# mapping cannot reach a process bound to container-internal loopback.
# ZTXBAS_CONSOLE_LISTEN_ADDR=127.0.0.1:8080

# TLS applies to both listeners when set.
# ZTXBAS_TLS_CERT=/path/to/cert.pem
# ZTXBAS_TLS_KEY=/path/to/key.pem

# ZTXBAS_JWT_SIGNING_KEY_PATH=./data/jwt-signing.key
# ZTXBAS_LOG_LEVEL=info

# --- Security: reverse-proxy trust ---
# CIDR list. When the direct peer belongs to one of these, X-Forwarded-For
# is honored for login rate limiting and audit-log source IP.
# ZTXBAS_TRUSTED_PROXIES=10.0.0.0/8,192.168.0.0/16

# --- Admin password policy ---
# Minimum admin password length. Floor 8, default 10.
# ZTXBAS_ADMIN_PW_MIN_LEN=10
# Force a password rotation after this many days. 0 disables the check.
# ZTXBAS_ADMIN_PW_MAX_AGE_DAYS=0
# Disable the bundled common-password deny list (testing only).
# ZTXBAS_ADMIN_PW_DENY_LIST_OFF=0

# --- Admin session pinning (off by default) ---
# Refuse to reuse a session from a different remote IP than the one it
# was issued with. Off by default because mobile-network roaming breaks it.
# ZTXBAS_ADMIN_SESSION_PIN_IP=0
# Refuse to reuse a session from a different User-Agent than the one it
# was issued with. Off by default because a browser update rewrites the UA
# string. Safer than IP pin for most deployments.
# ZTXBAS_ADMIN_SESSION_PIN_UA=0

# --- Console exposure fail-safe ---
# Startup refuses to bind the console to a non-loopback address without
# TLS unless this is explicitly set to 1. In Docker, set to 1 when the
# host-side port map (-p 127.0.0.1:8080:8080) already restricts host
# exposure to loopback — the container bind must be 0.0.0.0 for the
# mapping to work, so the fail-safe needs this ack. Do NOT set outside
# Docker unless you have TLS in front of the console.
# ZTXBAS_ALLOW_INSECURE_CONSOLE=0

# --- Audit-log retention ---
# Delete admin_audit rows older than this many days on the console sweeper
# cycle. 0 = keep forever.
# ZTXBAS_AUDIT_RETENTION_DAYS=0

# NOTE: the initial admin account is auto-created on first boot with a
# random password printed to stdout. Use the breakglass password
# to log in, change it and then rotate through the console.
